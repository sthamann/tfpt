namespace RHProofSearch
def arch_weighted : Prop := RH.SelectedArchWeightedInterpolationEstimate
def arch_rate : Prop := RH.SelectedArchErrorQuadraticRateExists
def arch_convergence : Prop := ∀ f : RH.GridElement, Filter.Tendsto (fun k : ℕ => RH.selectedArchError k f) Filter.atTop (nhds 0)
def freq : Prop := RH.frequently_selected_augDualResolvent_ge_half
def eventual_bound : Prop := RHSeparatedTail20260905.EventuallySelectedReadLowerBound
def native_positive : Prop := ∀ f : RH.GridElement, 0 ≤ RH.weilForm f
def gabor_positive : Prop := ∀ (a ω : ℚ) (ha : 0 < (a : ℝ)), 0 ≤ RH.gaborPoleSide (RH.pureGaborTest a ω ha) - RH.gaborPrimeComb (RH.pureGaborTest a ω ha) + RH.gaborArchSide (RH.pureGaborTest a ω ha)
def rh : Prop := RiemannHypothesis
theorem arch_weighted_proved  : arch_weighted := by
  exact RHArchRateClosure20260905.selectedArchWeightedInterpolationEstimate_holds
theorem arch_rate_from_weighted (h0 : arch_weighted) : arch_rate := by
  exact RH.selectedArchErrorQuadraticRateExists_of_weightedInterpolationEstimate h0
theorem arch_convergence_from_rate (h0 : arch_rate) : arch_convergence := by
  exact RH.selectedArchError_tendsto_zero_of_rateExists h0
theorem native_from_freq_and_bound (h0 : freq) (h1 : eventual_bound) : native_positive := by
  exact (RHFrontierClosure20260905.eventual_bound_iff_weil_nonneg h0).mp h1
theorem bound_from_freq_and_native (h0 : freq) (h1 : native_positive) : eventual_bound := by
  exact (RHFrontierClosure20260905.eventual_bound_iff_weil_nonneg h0).mpr h1
theorem rh_from_gabor (h0 : gabor_positive) : rh := by
  exact RH.gabor_primeSide_rational_criterion_iff_rh.mp h0
theorem gabor_from_rh (h0 : rh) : gabor_positive := by
  exact RH.gabor_primeSide_rational_criterion_iff_rh.mpr h0
end RHProofSearch
