000012 | \begin{definition}[Full admissibility complex]
000013 | \label{def:full-admissibility-complex}
000014 | Once the discrete carrier and determinant data are fixed on the admissible branch, the Hilbert
000015 | space carries a color factor and a determinant involution. Write
000016 | \[
000017 | \mathcal H=\mathcal H_{\Seam}\otimes\mathcal H_c\otimes\mathcal H_\Theta\otimes\mathcal H_t,
000018 | \qquad
000019 | \mathcal H_{\mathrm{ext}}
000020 | :=
000021 | \mathcal H\otimes\Lambda(\eta_\Sigma,\eta_c,\eta_\Theta,\eta_g),
000022 | \]
000023 | with carrier-side color generator $Z_c$ and determinant-sector involution $Q_\Theta$ satisfying
000024 | \[
000025 | \iC^2=\mathbf 1,
000026 | \qquad
000027 | Q_\Theta^2=\mathbf 1,
000028 | \qquad
000029 | Q_\Theta^\dagger=Q_\Theta.
000030 | \]
000031 | Define the full structural selector
000032 | \[
000033 | P_{\mathrm{str}}:=P_{\Seam,+}P_{\mathrm{sing}}P_\Theta,
000034 | \qquad
000035 | P_{\Seam,+}:=\frac{\mathbf 1+\iC}{2},
000036 | \qquad
000037 | P_\Theta:=\frac12(\mathbf 1+Q_\Theta),
000038 | \]
000039 | \[
000040 | P_{\mathrm{sing}}:=\frac13(\mathbf 1+Z_c+Z_c^2).
000041 | \]
000042 | Let $\epsilon_{\mathrm{adm}}>0$ and let $D_{\mathrm{adm}}$ be the admissible sector operator on
000043 | $\mathrm{Ran}(P_{\mathrm{str}})$ commuting with $\iC$, $P_{\mathrm{sing}}$, and $Q_\Theta$.
000044 | Define the elementary penalties
000045 | \[
000046 | K_\Sigma:=\sqrt{\lambda_\Sigma}\,\frac{\mathbf 1-\iC}{2},
000047 | \qquad
000048 | K_c:=\sqrt{\lambda_c}\,\bigl(\mathbf 1-P_{\mathrm{sing}}\bigr),
000049 | \]
000050 | \[
000051 | K_\Theta:=\sqrt{\lambda_\Theta}\,\frac{\mathbf 1-Q_\Theta}{2},
000052 | \qquad
000053 | K_g:=\sqrt{\lambda_{\mathrm{gap}}}\,\mathbf 1_{[0,\epsilon_{\mathrm{adm}}^2/4)}\!\bigl(P_{\mathrm{str}}D_{\mathrm{adm}}^\dagger D_{\mathrm{adm}}P_{\mathrm{str}}\bigr).
000054 | \]
000055 | The full admissibility differential is
000056 | \[
000057 | Q_{\mathrm{adm}}
000058 | :=
000059 | \eta_\Sigma^\dagger K_\Sigma
000060 | +\eta_c^\dagger K_c
000061 | +\eta_\Theta^\dagger K_\Theta
000062 | +\eta_g^\dagger K_g,
000063 | \]
000064 | and the full admissibility Laplacian is
000065 | \[
000066 | \Delta_{\mathrm{adm}}:=\{Q_{\mathrm{adm}},Q_{\mathrm{adm}}^\dagger\}.
000067 | \]
000068 | \end{definition}

000070 | \begin{theorem}[Two-stage upgrade from $\Pprim$ to $\Padm$]
000071 | \label{thm:padm-upgrade}
000072 | On the closed branch obtained after the carrier theorem and the compact bosonic index closure,
000073 | \[
000074 | Q_{\mathrm{adm}}^2=0,
000075 | \qquad
000076 | \Delta_{\mathrm{adm}}=K_\Sigma^2+K_c^2+K_\Theta^2+K_g^2=\Kadm,
000077 | \]
000078 | and the full Hodge projector
000079 | \[
000080 | \Padm
000081 | :=
000082 | \Pi_{\ker\Delta_{\mathrm{adm}}}
000083 | =
000084 | \operatorname*{s-lim}_{\beta\to\infty}e^{-\beta\Kadm}
000085 | \]
000086 | admits the two-stage factorization
000087 | \[
000088 | \Padm
000089 | =
000090 | \Pprim\,P_{\mathrm{sing}}\,P_\Theta
000091 | =
000092 | P_{\mathrm{str}}\,P_{\mathrm{gap}},
000093 | \qquad
000094 | P_{\mathrm{gap}}
000095 | :=
000096 | \mathbf 1_{[\epsilon_{\mathrm{adm}}^2/4,\infty)}\!\bigl(P_{\mathrm{str}}D_{\mathrm{adm}}^\dagger D_{\mathrm{adm}}P_{\mathrm{str}}\bigr).
000097 | \]
000098 | The first factor $\Pprim$ depends only on the primitive datum
000099 | $(\iC,\APSdom,D_{\mathrm{rel}})$; the second factor $P_{\mathrm{sing}}P_\Theta$ depends only on
000100 | the carrier and determinant data made available by the carrier and bosonic index theorems.
000101 | \end{theorem}

000103 | \begin{proof}[Proof sketch]
000104 | The auxiliary generators $\eta_\Sigma^\dagger,\eta_c^\dagger,\eta_\Theta^\dagger,\eta_g^\dagger$
000105 | anticommute, so $Q_{\mathrm{adm}}^2$ reduces to the commutators of $K_\Sigma,K_c,K_\Theta,K_g$.
000106 | The four commuting sectors $\mathcal H_\Seam,\mathcal H_c,\mathcal H_\Theta,\mathcal H_t$ are
000107 | acted on independently by these penalties, hence the mixed terms vanish and $\Delta_{\mathrm{adm}}=\Kadm$. The strong-coupling limit therefore projects onto the harmonic
000108 | admissible sector. Because $K_\Sigma$ and $K_g$ already define $\Pprim$ on
000109 | $\mathcal H_{\mathrm{prim}}$ and the additional penalties $K_c,K_\Theta$ commute with both, the
000110 | full projector factorizes through $\Pprim$ multiplicatively, giving the two-stage form.
000111 | \end{proof}

000113 | \begin{corollary}[Factorized selector as a derived form]
000114 | \label{cor:factorized-admissibility-selector}
000115 | On the admissible branch the familiar factorized selector
000116 | \[
000117 | \Padm=P_{\Seam,+}P_{\mathrm{sing}}P_\Theta\,P_{\mathrm{gap}}
000118 | \]
000119 | is recovered as a corollary of \TFPTxref{thm:padm-upgrade}; nothing in the primitive section depends
