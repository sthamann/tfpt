000001 | \subsection{Primitive relative spectral dynamics}
000002 | 
000003 | The carrier theorem is not intended to float above an unspecified microscopic dynamics. The
000004 | starting point is therefore the one-sided boundary datum from which the primitive layer, the
000005 | derived closed datum, and the canonical admissibility projector are reconstructed before the later
000006 | closure theorems are invoked.
000007 | 
000008 | \begin{definition}[One-sided boundary admissibility datum]
000009 | \label{def:reduced-theorem-datum}
000010 | The one-sided boundary admissibility datum is
000011 | \[
000012 | \Tbound:=(\mathcal A_+,\mathcal H_+,D_+,J,\Gamma,B_\Sigma)
000013 | \]
000014 | with the following built-in admissibility conditions:
000015 | \begin{enumerate}[label=(\roman*), leftmargin=1.5em, topsep=3pt, itemsep=2pt]
000016 |     \item $Z(\mathcal A_+)$ is spectrally regular and its Gelfand spectrum is a smooth compact oriented manifold with boundary $M_+$;
000017 |     \item $B_\Sigma$ is a self-adjoint tangential boundary operator on $\Sigma:=\partial M_+$ compatible with the boundary restrictions of $J$ and $\Gamma$;
000018 |     \item in a collar neighborhood of $\Sigma$, the Dirac operator has canonical product form
000019 |     \[
000020 |     D_+=\gamma_n(\partial_n+B_\Sigma);
000021 |     \]
000022 |     \item the boundary collar is reflection-regular so that canonical doubling, the Calder\'on projector, and the admissibility complex are well posed on the same datum.
000023 | \end{enumerate}
000024 | \end{definition}
000025 | 
000026 | \begin{remark}[Almost-commutative realizations]
000027 | \label{ass:almost-commutative-factorization}
000028 | When a factorized realization is useful, one may choose
000029 | \[
000030 | \mathcal H\cong L^2(\widetilde M,S)\otimes \Hint,
000031 | \]
000032 | with $S\to\widetilde M$ the spinor bundle on the oriented double cover and $\Hint$ a finite
000033 | internal Hilbert space. This is an analytic realization choice rather than part of the boundary
000034 | datum.
000035 | \end{remark}

000037 | \begin{theorem}[Doubling, deck involution, and boundary polarization]
000038 | \label{thm:primitive-datum-from-reduced-datum}
000039 | \label{thm:primitive-scaffold-from-reduced-datum}
000040 | Let $\Tbound$ be as above. Let
000041 | \[
000042 | \widetilde M:=M_+\cup_{\Sigma} M_+^{\mathrm{op}}
000043 | \]
000044 | be the doubled manifold and let
000045 | \[
000046 | \taudbl:\widetilde M\to\widetilde M
000047 | \]
000048 | be its deck involution. Let $P_C$ be the Calder\'on projector of the APS-realized boundary problem
000049 | for $D_+$ on $\Sigma$, and define the boundary polarization operator
000050 | \[
000051 | \iC:=2P_C-1
000052 | \]
000053 | on the Cauchy-data Hilbert space $\mathcal H_{\partial}$.
000054 | 
000055 | Then:
000056 | \begin{enumerate}[label=(\roman*), leftmargin=1.8em, topsep=3pt, itemsep=2pt]
000057 |     \item the geometric objects are
000058 |     \[
000059 |     M=\widetilde M/\taudbl,
000060 |     \qquad
000061 |     \Seam=\mathrm{Fix}(\taudbl),
000062 |     \qquad
000063 |     \Xbulk=\widetilde M\setminus\Seam;
000064 |     \]
000065 |     \item the doubled operator and its even and odd parts are
000066 |     \[
000067 |     D=D_+\cup_{\Sigma}D_+^{\mathrm{op}},
000068 |     \qquad
000069 |     D_{\mathrm{ref}}:=\frac12\bigl(D+\taudbl^{\ast}D\taudbl^{\ast}\bigr),
000070 |     \qquad
000071 |     D_{\mathrm{rel}}:=\frac12\bigl(D-\taudbl^{\ast}D\taudbl^{\ast}\bigr);
000072 |     \]
000073 |     \item on boundary Cauchy data the induced reflection $R_{\taudbl}$ satisfies
000074 |     \[
000075 |     \operatorname{Ran}(P_C)=\ker(R_{\taudbl}-1),
000076 |     \qquad
000077 |     \ker(P_C)=\ker(R_{\taudbl}+1),
000078 |     \qquad
000079 |     \iC=R_{\taudbl}.
000080 |     \]
000081 | \end{enumerate}
000082 | Hence the primitive scaffold
000083 | \[
000084 | \Tprim^{\mathrm{kin}}
000085 | :=
000086 | (\Xbulk,\widetilde M,M,\Seam,\taudbl,\iC,D_{\mathrm{ref}},D_{\mathrm{rel}},\chiseed)
000087 | \]
000088 | is derived, not assumed. The derived doubled datum may be recorded as
000089 | \[
000090 | \Tmin^{\mathrm{cl}}:=(\mathcal A,\mathcal H,D,J,\Gamma,\taudbl,\iC),
000091 | \]
000092 | If one further chooses an almost-commutative factorization
000093 | \[
000094 | \mathcal H_{\mathrm{prim}}\cong L^2(\Xbulk,S)\otimes \Hint,
000095 | \]
000096 | then $S$ and $\Hint$ refer only to that chosen realization and are not additional
000097 | reconstruction outputs.
000098 | \end{theorem}
000099 | 
000100 | \begin{proof}
000101 | Reflection regularity on an exact collar produces the doubled manifold together with the deck
000102 | involution $\taudbl$. The APS-realized boundary problem has a Calder\'on projector $P_C$ whose
000103 | range is the Cauchy-data space of interior solutions. On the doubled collar the deck reflection
000104 | acts on Cauchy data by $R_{\taudbl}$, and the solution space splits into the $\pm1$ eigenspaces of
000105 | this reflection. Therefore
000106 | \[
000107 | P_C=\frac{1+R_{\taudbl}}{2},
000108 | \qquad
000109 | \iC=2P_C-1=R_{\taudbl}.
000110 | \]
000111 | The formulas for $D_{\mathrm{ref}}$ and $D_{\mathrm{rel}}$ are the even and odd parts of $D$ under
000112 | the induced involution.
000113 | \end{proof}
000114 | 
000115 | \begin{corollary}[No confusion among the three involutions]
000116 | Geometric reflection is carried by $\taudbl$, boundary polarization by $\iC$, and the finite
000117 | carrier split by the restricted involution
000118 | \[
000119 | \epscar:=\iC|_{\mathcal H_{\mathrm{car}}}.
000120 | \]
000121 | No later theorem may identify these three operators literally; only the displayed restriction and
000122 | transport maps are allowed.
000123 | \end{corollary}
