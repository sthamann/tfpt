000098 | def fixed_components_of_real_involution(f: Affine) -> int:
000099 |     """Count circles in Fix(f) by the rank-one Smith invariant of I-M.
000100 | 
000101 |     For (I-M)v=d mod Z^2, a primitive integer left-null vector tests
000102 |     solvability.  If solvable, gcd of the entries of I-M is the component
000103 |     group order of its torus kernel, hence the number of fixed circles.
000104 |     """
000105 |     matrix, shift = f
000106 |     if determinant(matrix) != -1 or element_order(f) != 2:
000107 |         raise ValueError("fixed-circle classification requires a real involution")
000108 |     a = (
000109 |         (1 - matrix[0][0], -matrix[0][1]),
000110 |         (-matrix[1][0], 1 - matrix[1][1]),
000111 |     )
000112 |     nonzero_column = next(
000113 |         (a[0][j], a[1][j]) for j in range(2) if (a[0][j], a[1][j]) != (0, 0)
000114 |     )
000115 |     divisor = gcd(abs(nonzero_column[0]), abs(nonzero_column[1]))
000116 |     left_null = (nonzero_column[1] // divisor, -nonzero_column[0] // divisor)
000117 |     obstruction = left_null[0] * shift[0] + left_null[1] * shift[1]
000118 |     if obstruction.denominator != 1:
000119 |         return 0
000120 |     entries = [abs(x) for row in a for x in row if x]
000121 |     return reduce(gcd, entries)
