000203 | def _sccs(graph):
000204 |     """Tarjan strongly connected components (iterative)."""
000205 |     idx, low, on, order = {}, {}, {}, []
000206 |     stack, out = [], []
000207 |     for root in graph:
000208 |         if root in idx:
000209 |             continue
000210 |         work = [(root, 0)]
000211 |         while work:
000212 |             v, pi = work.pop()
000213 |             if pi == 0:
000214 |                 idx[v] = low[v] = len(idx)
000215 |                 order.append(v)
000216 |                 on[v] = True
000217 |             recurse = False
000218 |             nbrs = graph.get(v, [])
000219 |             for i in range(pi, len(nbrs)):
000220 |                 w = nbrs[i]
000221 |                 if w not in idx:
000222 |                     work.append((v, i + 1))
000223 |                     work.append((w, 0))
000224 |                     recurse = True
000225 |                     break
000226 |                 if on.get(w):
000227 |                     low[v] = min(low[v], idx[w])
000228 |             if recurse:
000229 |                 continue
000230 |             if low[v] == idx[v]:
000231 |                 comp = []
000232 |                 while True:
000233 |                     w = order.pop()
000234 |                     on[w] = False
