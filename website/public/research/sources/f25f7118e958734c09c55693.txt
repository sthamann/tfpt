000065 | \begin{definition}[One-sided boundary admissibility datum]
000066 | \label{def:reduced-theorem-datum}
000067 | The one-sided boundary admissibility datum is
000068 | \[
000069 | \Tbound:=(\mathcal A_+,\mathcal H_+,D_+,J,\Gamma,B_\Sigma)
000070 | \]
000071 | with the following built-in admissibility conditions:
000072 | \begin{enumerate}[label=(\roman*), leftmargin=1.5em, topsep=3pt, itemsep=2pt]
000073 |     \item $Z(\mathcal A_+)$ is spectrally regular and its Gelfand spectrum is a smooth compact oriented manifold with boundary $M_+$;
000074 |     \item $B_\Sigma$ is a self-adjoint tangential boundary operator on $\Sigma:=\partial M_+$ compatible with the boundary restrictions of $J$ and $\Gamma$;
000075 |     \item in a collar neighborhood of $\Sigma$, the Dirac operator has canonical product form
000076 |     \[
000077 |     D_+=\gamma_n(\partial_n+B_\Sigma);
000078 |     \]
000079 |     \item the boundary collar is reflection-regular so that canonical doubling, the Calder\'on projector, and the admissibility complex are well posed on the same datum.
000080 | \end{enumerate}
000081 | \end{definition}

000083 | \begin{remark}[Almost-commutative realizations]
000084 | \label{ass:almost-commutative-factorization}
000085 | When a factorized realization is useful, one may choose

000087 | \mathcal H\cong L^2(\widetilde M,S)\otimes \Hint,
000088 | \]

000094 | \begin{theorem}[Doubling, deck involution, and boundary polarization]
000095 | \label{thm:primitive-datum-from-reduced-datum}
000096 | \label{thm:primitive-scaffold-from-reduced-datum}
000097 | Let $\Tbound$ be as above. Let
000098 | \[
000099 | \widetilde M:=M_+\cup_{\Sigma} M_+^{\mathrm{op}}
000100 | \]
000101 | be the doubled manifold and let
000102 | \[
000103 | \taudbl:\widetilde M\to\widetilde M
000104 | \]
000105 | be its deck involution. Let $P_C$ be the Calder\'on projector of the APS-realized boundary problem
000106 | for $D_+$ on $\Sigma$, and define the boundary polarization operator
000107 | \[
000108 | \iC:=2P_C-1
000109 | \]
000110 | on the Cauchy-data Hilbert space $\mathcal H_{\partial}$.
000111 | 
000112 | Then:
000113 | \begin{enumerate}[label=(\roman*), leftmargin=1.8em, topsep=3pt, itemsep=2pt]
000114 |     \item the geometric objects are
000115 |     \[
000116 |     M=\widetilde M/\taudbl,
000117 |     \qquad
000118 |     \Seam=\mathrm{Fix}(\taudbl),
000119 |     \qquad
000120 |     \Xbulk=\widetilde M\setminus\Seam;
000121 |     \]
000122 |     \item the doubled operator and its even and odd parts are
000123 |     \[
000124 |     D=D_+\cup_{\Sigma}D_+^{\mathrm{op}},
000125 |     \qquad
000126 |     D_{\mathrm{ref}}:=\frac12\bigl(D+\taudbl^{\ast}D\taudbl^{\ast}\bigr),
000127 |     \qquad
000128 |     D_{\mathrm{rel}}:=\frac12\bigl(D-\taudbl^{\ast}D\taudbl^{\ast}\bigr);
000129 |     \]
000130 |     \item on boundary Cauchy data the induced reflection $R_{\taudbl}$ satisfies
000131 |     \[
000132 |     \operatorname{Ran}(P_C)=\ker(R_{\taudbl}-1),
000133 |     \qquad
000134 |     \ker(P_C)=\ker(R_{\taudbl}+1),
000135 |     \qquad
000136 |     \iC=R_{\taudbl}.
000137 |     \]
000138 | \end{enumerate}
000139 | Hence the primitive scaffold
000140 | \[
000141 | \Tprim^{\mathrm{kin}}
000142 | :=
000143 | (\Xbulk,\widetilde M,M,\Seam,\taudbl,\iC,D_{\mathrm{ref}},D_{\mathrm{rel}},\chiseed)
000144 | \]
000145 | is derived, not assumed. The derived doubled datum may be recorded as
000146 | \[
000147 | \Tmin^{\mathrm{cl}}:=(\mathcal A,\mathcal H,D,J,\Gamma,\taudbl,\iC),
000148 | \]
000149 | If one further chooses an almost-commutative factorization
000150 | \[
000151 | \mathcal H_{\mathrm{prim}}\cong L^2(\Xbulk,S)\otimes \Hint,
000152 | \]
000153 | then $S$ and $\Hint$ refer only to that chosen realization and are not additional
000154 | reconstruction outputs.
000155 | \end{theorem}
