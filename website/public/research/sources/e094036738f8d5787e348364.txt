000047 | \begin{theorem}[Charged MMST master theorem: exact proved form]
000048 | \label{thm:master}
000049 | Let the finite collars, scaling maps, neutral current limit and charged
000050 | operators be as in Section~\ref{sec:setup}.  Suppose the scaling algebra
000051 | is exhausted by the neutral currents and
000052 | the four charged grades as specified in \textup{(ALG-EXH)} there.  Then:
000053 | \begin{enumerate}[label=\textup{(L\arabic*)}]
000054 | \item For every $f\in C^\infty(S^1)$ there are $p<\infty$ and
000055 |   $C_f<\infty$, independent of $N$, such that on the finite-energy core
000056 |   \[
000057 |     \norm{\Psi_{\lambda,N}(f)\xi}
000058 |     \le C_f\norm{(1+L_{0,N})^p\xi}.
000059 |   \tag{1.1}\label{eq:L1}

000061 | \item After the canonical scaling identifications,
000062 |   $\Psi_{\lambda,N}(f)$ converges strongly on the common finite-energy
000063 |   core to a nonzero closable field $\Psi_\lambda(f)$.  The generated local
000064 |   algebras converge to a local, conformal extension net
000065 |   $I\mapsto\B(I)$ of the neutral net $I\mapsto\A(I)$.

000068 |   inductive/scaling limit.  The limiting Pimsner--Popa bound is still
000069 |   $E(x)\ge\frac14x$ for $x\ge0$.
