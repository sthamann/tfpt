000144 | \begin{definition}[Admissible Schwinger functions]
000145 | \label{def:admissible-schwinger-functions}
000146 | Let
000147 | \[
000148 | W_{\mathrm{rel}}[J,\eta,\bar\eta]:=\log Z_{\mathrm{rel}}[J,\eta,\bar\eta].
000149 | \]
000150 | For local source insertions supported in the Euclidean continuation of
000151 | $\mathcal U_{\mathrm{scat}}$, define the truncated admissible Schwinger distributions by
000152 | \[
000153 | S_n^T(x_1,\ldots,x_n)
000154 | :=
000155 | \left.
000156 | \frac{\delta^n W_{\mathrm{rel}}}{\delta\mathcal J(x_1)\cdots\delta\mathcal J(x_n)}
000157 | \right|_{\mathcal J=0},
000158 | \qquad
000159 | \mathcal J:=(J,\eta,\bar\eta).

000163 | \begin{theorem}[\textnormal{[M]} Full admissible Osterwalder--Schrader closure]
000164 | \label{thm:full-admissible-os-closure}
000165 | On the Euclidean continuation of $\mathcal U_{\mathrm{scat}}$, the family
000166 | $\{S_n^T\}_{n\ge 1}$ satisfies the local Osterwalder--Schrader axioms: regularity, Euclidean
000167 | covariance, reflection positivity, symmetry, and clustering. Hence there exists a reconstructed
000168 | Wightman theory
000169 | \[
000170 | (\mathcal H_{\mathrm{OS}},U(a,\Lambda),\Omega,\{\Phi_i\})
000171 | \]
000172 | whose Schwinger functions are the boundary values of the admissible Euclidean family
000173 | $\{S_n^T\}_{n\ge 1}$.
000174 | \end{theorem}
000175 | 
000176 | \begin{proof}
000177 | Regularity and moment bounds are recorded in
000178 | \Cref{lem:tempered-schwinger-bounds}, while graded Euclidean covariance and symmetry on $\Padm$ are
000179 | recorded in \Cref{lem:graded-euclidean-covariance-padm}. Reflection positivity is exactly
000180 | \TFPTxref{thm:reflection-positivity-admissible}, and clustering is the exponential clustering statement
000181 | of \TFPTxref{thm:nonperturbative-admissible-qft}. Therefore the admissible Schwinger family satisfies
000182 | the hypotheses of the tailored OS/CAR reconstruction theorem
000183 | \Cref{thm:tailored-os-car-reconstruction}, which yields the claimed reconstructed Wightman theory.
000184 | \end{proof}

000186 | \begin{theorem}[Admissible local Minkowski net and microcausality]
000187 | \label{thm:admissible-local-minkowski-net}
000188 | Let $\mathfrak A_{\mathrm{adm}}(\mathcal O)$ be the physical admissible local net of
000189 | \TFPTxref{def:physical-admissible-local-net}. Then:
000190 | \begin{enumerate}[label=\textup{(\roman*)}, leftmargin=1.8em, topsep=3pt, itemsep=2pt]
000191 |     \item if $\mathcal O_1\subset\mathcal O_2$, then
000192 |     \[
000193 |     \mathfrak A_{\mathrm{adm}}(\mathcal O_1)\subset \mathfrak A_{\mathrm{adm}}(\mathcal O_2);
000194 |     \]
000195 |     \item if $\mathcal O_1\perp\mathcal O_2$, then
000196 |     \[
000197 |     [A_1,A_2]=0
000198 |     \qquad
000199 |     (A_1\in\mathfrak A_{\mathrm{adm}}(\mathcal O_1),\ A_2\in\mathfrak A_{\mathrm{adm}}(\mathcal O_2)).
000200 |     \]
000201 | \end{enumerate}
000202 | Equivalently, the admissible branch reconstructs a causal Minkowski local net on
000203 | $\mathcal H_{\mathrm{adm}}$.
000204 | \end{theorem}
000205 | 
000206 | \begin{proof}
000207 | Isotony is immediate from the support definition of the unreduced local net
000208 | $\mathfrak A_M(\mathcal O)$. For locality, consider the Wick-rotated quadratic operators on the
000209 | low-curvature branch. Scalar and bosonic modes are normally hyperbolic, fermions are
000210 | Dirac-hyperbolic, and gauge fields become hyperbolic after covariant gauge fixing and BRST
000211 | reduction. Hence the advanced and retarded fundamental solutions $E^\pm$ exist and satisfy
000212 | \[
000213 | \operatorname{supp}(E^\pm)\subset J^\pm.
000214 | \]
000215 | Therefore the Pauli--Jordan propagator
000216 | \[
000217 | E:=E^+-E^-
000218 | \]
000219 | has support in the causal cone, so for spacelike separated test-function supports one has
000220 | \[
000221 | [\Phi_i(f),\Phi_j(g)]_{\mp}=0.
000222 | \]
000223 | Now let $A_1$ and $A_2$ be local operators preserving $\mathcal H_{\mathrm{adm}}$. Since they
000224 | commute on $\mathcal H_{\mathrm{OS}}$ for spacelike separated supports, their restrictions to the
000225 | invariant subspace $\mathcal H_{\mathrm{adm}}$ also commute:
000226 | \[
000227 | [A_1|_{\mathcal H_{\mathrm{adm}}},A_2|_{\mathcal H_{\mathrm{adm}}}]=0.
000228 | \]
000229 | Thus locality survives admissibility because admissibility is implemented as a state-space
000230 | restriction, not as a nonlocal compression defining the algebra itself.
000231 | \end{proof}
